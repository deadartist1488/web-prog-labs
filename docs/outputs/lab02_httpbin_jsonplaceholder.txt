# Занятие 3, задания 7 и 9: методы HTTP на httpbin и JSONPlaceholder

$ curl -X GET https://httpbin.org/get
HTTP 200; эхо: method/url = GET, json = None, args = {}

$ curl -X POST https://httpbin.org/post
HTTP 200; эхо: method/url = POST, json = {'age': 30, 'name': 'John'}, args = {}

$ curl -X PUT https://httpbin.org/put
HTTP 200; эхо: method/url = PUT, json = {'age': 30, 'name': 'John'}, args = {}

$ curl -X PATCH https://httpbin.org/patch
HTTP 200; эхо: method/url = PATCH, json = {'age': 30, 'name': 'John'}, args = {}

$ curl -X DELETE https://httpbin.org/delete
HTTP 200; эхо: method/url = DELETE, json = {'age': 30, 'name': 'John'}, args = {}

$ curl https://jsonplaceholder.typicode.com/posts
HTTP 200
100 постов, ключи: ['userId', 'id', 'title', 'body']

$ curl https://jsonplaceholder.typicode.com/posts/1

  "userId": 1,
  "id": 1,
